# Planificador: maximo MAX agentes, reintentos, arranque escalonado, respeta la RAM libre.
C=$HOME/untitled-control/cycle2; LOG=$HOME/untitled-control/sched.log
MAX=${MAX:-2}
# tarea:dependencia:pesada(1=Gradle/JVM)
QUEUE="I1_aiworld:: K1_carsworld:I1_aiworld: A1_realaudio:: N1_animation:: D1_datadriven:: D2_modevents:D1_datadriven: D3_modtools:D2_modevents:"
declare -A tries; last=0
log(){ echo "$(date +%T) $*" | tee -a $LOG; }
swapused(){ awk '/SwapTotal/ {t=$2} /SwapFree/ {f=$2} END {print int((t-f)/1024)}' /proc/meminfo; }
maxnow(){ if pgrep -f "title C2_[V]1_vrbuild" >/dev/null; then echo 2; elif [ "$(swapused)" -lt 2800 ] && [ "$(avail)" -ge 1800 ]; then echo 3; else echo 2; fi; }
avail(){ awk '/MemAvailable/ {print int($2/1024)}' /proc/meminfo; }
isdone(){ [ -f $C/$1.exit ] && { [ "$(cat $C/$1.exit)" = 0 ] || [ "$(cat $C/$1.exit)" = 99 ]; }; }
running(){ pgrep -f "title C2_$1" >/dev/null; }
log "sched start MAX=$MAX avail=$(avail)MB"
while true; do
  run=0; pending=0
  for e in $QUEUE; do t=${e%%:*}; running $t && run=$((run+1)); done
  drain=0
  for e in $QUEUE; do IFS=: read t dep heavy <<<"$e"; [ "$heavy" = 1 ] && ! isdone $t && ! running $t && { [ -z "$dep" ] || isdone $dep; } && drain=1; done
  for e in $QUEUE; do
    IFS=: read t dep heavy <<<"$e"
    isdone $t && continue
    running $t && continue
    pending=$((pending+1))
    [ -n "$dep" ] && ! isdone $dep && continue
    [ $run -ge $(maxnow) ] && break
    [ $drain = 1 ] && [ "$heavy" != 1 ] && continue
    need=1500; [ "$heavy" = 1 ] && need=1900
    pgrep -f "title C2_[V]1_vrbuild" >/dev/null && need=1400
    [ "$heavy" = 1 ] && [ $run -gt 1 ] && continue
    [ "$(avail)" -lt $need ] && break
    [ "$(swapused)" -gt 3400 ] && break
    pgrep -f "org.gradle|[g]radle-launcher|GradleDaemon" >/dev/null && break
    [ $(( $(date +%s) - last )) -lt 45 ] && break

L=$HOME/untitled-control/locks; mkdir -p "$L"
while true; do
  for i in 1 2; do
    exec {fd}>"$L/godot$i.lock"

  done
